cmake_minimum_required(VERSION 3.28)
project(gpu_nci VERSION 0.1.0 LANGUAGES C Fortran)
include(CTest)
option(NCI_ENABLE_OPENMP "Enable CPU OpenMP" ON)
option(NCI_BUILD_TEST_HOOKS "Build separate test-only numerical hooks" OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  add_compile_options("$<$<COMPILE_LANGUAGE:Fortran>:-ffree-line-length-none;-ffp-contract=off;-fimplicit-none>")
  add_compile_options("$<$<AND:$<COMPILE_LANGUAGE:Fortran>,$<CONFIG:Debug>>:-fcheck=all;-fbacktrace;-Wall;-Wextra>")
elseif(CMAKE_Fortran_COMPILER_ID STREQUAL "NVHPC")
  add_compile_options("$<$<COMPILE_LANGUAGE:Fortran>:-Kieee;-Mnofma>")
  add_compile_options("$<$<AND:$<COMPILE_LANGUAGE:Fortran>,$<CONFIG:Debug>>:-Mbounds>")
endif()
if(NCI_ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS Fortran)
endif()
add_library(nci_contract INTERFACE)
target_include_directories(nci_contract INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(nci_abi_layout tests/abi/layout.c)
target_link_libraries(nci_abi_layout PRIVATE nci_contract)
add_test(NAME abi_c_layout COMMAND nci_abi_layout)
set(NCI_CPU_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/backend_cpu" CACHE PATH "CPU source directory; external worktrees may be tested without merging")
set(NCI_IO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/io" CACHE PATH "Input source directory")
set(NCI_ORACLE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Independent oracle checkout")
set(NCI_DRIVER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/driver" CACHE PATH "Native CLI worker source directory")
set(NCI_CLI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cli" CACHE PATH "Python CLI directory")
set(NCI_CLI_TEST_DIR "${NCI_CLI_DIR}/../tests/cli" CACHE PATH "CLI unit tests directory")
set(NCI_E2E_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tests/e2e" CACHE PATH "Independent CLI end-to-end tests")
set(NCI_OPENACC_PROBE_TEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tests/openacc_probe" CACHE PATH "Pure independent probe verifier selfchecks; never executes a GPU")
set(NCI_OPENACC_RESOURCE_TEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tests/openacc_resources" CACHE PATH "Pure independent resource verifier selfchecks; never executes a GPU")
set(NCI_BUILD_REVISION "UNKNOWN" CACHE STRING "Explicit source revision declaration; UNKNOWN unless supplied by builder")
if(EXISTS "${NCI_CPU_SOURCE_DIR}/CMakeLists.txt")
  add_subdirectory("${NCI_CPU_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cpu")
endif()
if(EXISTS "${NCI_IO_SOURCE_DIR}/CMakeLists.txt")
  add_subdirectory("${NCI_IO_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/io")
endif()
if(EXISTS "${NCI_DRIVER_SOURCE_DIR}/CMakeLists.txt")
  add_subdirectory("${NCI_DRIVER_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/driver")
endif()
if(BUILD_TESTING AND TARGET nci_driver AND EXISTS "${NCI_E2E_DIR}/CMakeLists.txt")
  add_subdirectory("${NCI_E2E_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/e2e")
endif()
if(BUILD_TESTING AND EXISTS "${NCI_CLI_TEST_DIR}/test_cli.py")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME offline_cli_unit
    COMMAND "${CMAKE_COMMAND}" -E env "NCI_CLI=${NCI_CLI_DIR}/nci_cli.py"
      "${Python3_EXECUTABLE}" -m pytest -q -ra "${NCI_CLI_TEST_DIR}")
  set_tests_properties(offline_cli_unit PROPERTIES TIMEOUT 180)
endif()
if(EXISTS "${NCI_ORACLE_DIR}/tests/oracle")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(TARGET nci_cpu)
    add_test(NAME independent_oracle
      COMMAND "${CMAKE_COMMAND}" -E env "NCI_LIBRARY=$<TARGET_FILE:nci_cpu>"
        "${Python3_EXECUTABLE}" -m pytest -q -ra "${NCI_ORACLE_DIR}/tests/oracle")
    set_tests_properties(independent_oracle PROPERTIES TIMEOUT 180)
  else()
    add_test(NAME reference_selfchecks_only
      COMMAND "${CMAKE_COMMAND}" -E env --unset=NCI_LIBRARY
        "${Python3_EXECUTABLE}" -m pytest -q -ra "${NCI_ORACLE_DIR}/tests/oracle")
    set_tests_properties(reference_selfchecks_only PROPERTIES TIMEOUT 180)
  endif()
endif()
if(BUILD_TESTING AND EXISTS "${NCI_OPENACC_PROBE_TEST_DIR}/test_verify.py")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME openacc_probe_verifier_selfchecks
    COMMAND "${Python3_EXECUTABLE}" -m pytest -q -ra "${NCI_OPENACC_PROBE_TEST_DIR}")
  set_tests_properties(openacc_probe_verifier_selfchecks PROPERTIES TIMEOUT 120)
endif()
if(BUILD_TESTING AND EXISTS "${NCI_OPENACC_RESOURCE_TEST_DIR}/test_verify.py")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME openacc_resource_verifier_selfchecks
    COMMAND "${Python3_EXECUTABLE}" -m pytest -q -ra "${NCI_OPENACC_RESOURCE_TEST_DIR}")
  set_tests_properties(openacc_resource_verifier_selfchecks PROPERTIES TIMEOUT 120)
endif()
